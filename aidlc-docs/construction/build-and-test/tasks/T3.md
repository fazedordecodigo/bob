## Especificação Mecânica das Alterações

### 1. `tests/test_security_mutants.py`

Localize a função `test_detached_worktree_names_the_directory_and_removes_it` (linhas 60–113):

```python
def test_detached_worktree_names_the_directory_and_removes_it(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    cwd = tmp_path / "repo"
    cwd.mkdir()
    calls: list[tuple[Path, tuple[str, ...]]] = []
    removed: list[tuple[Path, bool]] = []
    created: list[str] = []
    real_mkdtemp = security_gate.tempfile.mkdtemp
    real_rmtree = security_gate.shutil.rmtree

    def spy_mkdtemp(*, prefix: str) -> str:
        assert prefix == "bob-semgrep-"
        path = real_mkdtemp(prefix=prefix)
        created.append(path)
        return path

    def spy_git(repo: Path, *args: str) -> subprocess.CompletedProcess[str]:
        calls.append((repo, args))
        if args[:2] == ("worktree", "add"):
            Path(args[3]).mkdir(parents=True, exist_ok=True)
        return subprocess.CompletedProcess(["git", *args], 0, "", "")

    def spy_rmtree(path: Path, *, ignore_errors: bool = False) -> None:
        removed.append((path, ignore_errors))
        real_rmtree(path, ignore_errors=ignore_errors)

    monkeypatch.setattr(security_gate.tempfile, "mkdtemp", spy_mkdtemp)
    monkeypatch.setattr(security_gate, "_git", spy_git)
    monkeypatch.setattr(security_gate.shutil, "rmtree", spy_rmtree)
    with security_gate._detached(cwd) as root:
        parent = Path(created[0])
        assert root == parent / "head"

    assert calls == [
        (cwd, ("worktree", "add", "--detach", str(parent / "head"), "HEAD")),
        (cwd, ("worktree", "remove", "--force", str(parent / "head"))),
    ]
    assert removed == [(parent, True)]
    assert not parent.exists()

    def failing_git(repo: Path, *args: str) -> subprocess.CompletedProcess[str]:
        return subprocess.CompletedProcess(["git", *args], 1, "", "fatal: bad HEAD\n")

    monkeypatch.setattr(security_gate, "_git", failing_git)
    yielded = None
    with pytest.raises(ToolError):
        with security_gate._detached(cwd) as root:
            yielded = root

    parent = Path(created[1])
    assert yielded is None
    assert removed[-1] == (parent, True)
    assert not parent.exists()
```

Substitua as duas atribuições `parent = Path(created[0])` (linha 89) e `parent = Path(created[1])` (linha 108) **exatamente** por `parent = Path(created[0]).resolve()` e `parent = Path(created[1]).resolve()`:

```python
    with security_gate._detached(cwd) as root:
        parent = Path(created[0]).resolve()
        assert root == parent / "head"

    assert calls == [
        (cwd, ("worktree", "add", "--detach", str(parent / "head"), "HEAD")),
        (cwd, ("worktree", "remove", "--force", str(parent / "head"))),
    ]
    assert removed == [(parent, True)]
    assert not parent.exists()

    def failing_git(repo: Path, *args: str) -> subprocess.CompletedProcess[str]:
        return subprocess.CompletedProcess(["git", *args], 1, "", "fatal: bad HEAD\n")

    monkeypatch.setattr(security_gate, "_git", failing_git)
    yielded = None
    with pytest.raises(ToolError):
        with security_gate._detached(cwd) as root:
            yielded = root

    parent = Path(created[1]).resolve()
    assert yielded is None
    assert removed[-1] == (parent, True)
    assert not parent.exists()
```

*(Nota: em Python `pathlib.Path.resolve(strict=False)`, mesmo na linha 108 após `shutil.rmtree` remover o diretório folha `created[1]`, o ancestral `/var` → `/private/var` existe e é resolvido identicamente a `_detached`, preservando todas as asserções no Linux e no macOS.)*
